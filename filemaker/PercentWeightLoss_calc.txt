// PercentWeightLoss - Obesity Management, field id 100, Text calc (unstored).
// The patient progress summary block (also feeds the new-patient prompt via
// HPISummary's "Weight Loss Goal progress:" section and layout display).
// v2 (2026-09-26): HPIHighestWeight (self-reported pre-enrollment peak) header
// line + from-the-high loss figure when present AND > Weightpounds.
// v3 (2026-10-03): last weigh-in now comes from ONE real WeighIn row via SQL
// (Weight IS NOT NULL, newest Date first) instead of positional fetches
// (GetNthRecord Count-1 / Last()), which broke when placeholder rows existed
// (dates without weights, future appointment rows) - the "last weigh-in was N
// days ago" footer could cite a stale date that disagreed with the weight shown.
// Weeks now computed from that row's date vs DateOfFirstVisit; current weight,
// progress %, goal math and footer all use the same row. Verified in production.

Let ( [
    ~peak  = If ( not IsEmpty ( HPIHighestWeight ) and GetAsNumber ( HPIHighestWeight ) > Weightpounds ;
                  GetAsNumber ( HPIHighestWeight ) ; "" ) ;

    // latest REAL weigh-in: date + weight from the same row
    ~row = ExecuteSQL (
        "SELECT \"Date\", \"Weight\" FROM \"WeighIn\"
         WHERE \"AccountNumber\" = ? AND \"Weight\" IS NOT NULL AND \"Date\" IS NOT NULL
         ORDER BY \"Date\" DESC FETCH FIRST 1 ROWS ONLY" ; "|" ; "" ; AccountNumber ) ;
    ~iso = If ( ~row ≠ "?" ; GetValue ( Substitute ( ~row ; "|" ; ¶ ) ; 1 ) ) ;
    ~lastD = If ( not IsEmpty ( ~iso ) ; Date ( Middle ( ~iso ; 6 ; 2 ) ; Middle ( ~iso ; 9 ; 2 ) ; Left ( ~iso ; 4 ) ) ) ;
    ~lastW = If ( ~row ≠ "?" ; GetAsNumber ( GetValue ( Substitute ( ~row ; "|" ; ¶ ) ; 2 ) ) ) ;
    ~nW = GetAsNumber ( ExecuteSQL ( "SELECT COUNT(*) FROM \"WeighIn\" WHERE \"AccountNumber\" = ? AND \"Weight\" IS NOT NULL" ; "" ; "" ; AccountNumber ) ) ;
    ~weeks = Truncate ( ( ~lastD - DateOfFirstVisit ) / 7 ; 0 )
] ;

// --- SECTION: HIGHEST REPORTED WEIGHT (only when valid) ---
If ( ~peak ≠ "" ;
    "Highest Reported Weight: " & ~peak & " lbs (self-reported, pre-enrollment)" & Char (13) ;
    "" ) &

"Starting Weight: " & Weightpounds & " lbs at BMI: " & BMI_Initial & " (" & DateOfFirstVisit & ")"
& Char (13) &
"Current Weight: " & ~lastW & " lbs.  BMI: " & CurrentBMI &

// --- SECTION: PROGRESS (Only shows after the first visit) ---
If ( ~nW > 1 ;
    Char (13) &
    Truncate ( ((Weightpounds - ~lastW) / Weightpounds) * 100 ; 1 ) & "% weight loss at " &
    ~weeks & " weeks." & Char (13) &
    "Down " & Weightpounds - ~lastW & " lbs so far." &
    If ( ~peak ≠ "" and ~peak > ~lastW ;
        Char (13) & "From reported high of " & ~peak & ": down " & ( ~peak - ~lastW ) & " lbs (" &
        Truncate ( ( ~peak - ~lastW ) / ~peak * 100 ; 1 ) & "% total)." ;
        "" )
    ; ""
) &

// --- SECTION: GOAL & PROJECTION (Always shows) ---
Char (13) &
"Target Weight: " & TargetWeight & " lbs (BMI = 25)" & Char (13) &
"Pounds to lose: " & (~lastW - TargetWeight) & " lbs" & Char (13) &
"Estimated time to goal: " & Truncate ( (~lastW - TargetWeight) / 1.5 ; 1 ) & " weeks (at 1.5 lbs/week)" &

Char (13) & Char (13) &

// --- SECTION: LAST VISIT FOOTER ---
If ( IsEmpty ( ~lastD ) ; "" ;
Let ( [
    daysAgo = Get ( CurrentDate ) - ~lastD ;
    daysText = Case (
        daysAgo = 0 ; "today" ;
        daysAgo = 1 ; "1 day ago" ;
        daysAgo > 1 ; daysAgo & " days ago"
    ) ;
    coloredDaysText = If ( daysAgo ≥ 42 ; TextColor ( daysText ; RGB (255 ; 0 ; 0 ) ) ; daysText )
] ;
    "Last weigh-in was " & coloredDaysText &
    If ( daysAgo ≠ 0 ; " on " & ~lastD ; "" )
) )

)
