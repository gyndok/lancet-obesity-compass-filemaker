// WeighIn helper fields v2 - 2026-10-03 - Obesity Management.
// All five previously used positional fetches (GetNthRecord(...; Count(...)-1) or
// Last(WeighIn::Weight)) that break when placeholder WeighIn rows exist (dates without
// weights, including future appointment rows). The date helpers were also off by one row,
// and CurrentWeight/TimeSinceLastWeighInDisplay could cite DIFFERENT rows (mismatched
// weight vs "days ago"). v2: everything derives from the latest real row (Weight IS NOT
// NULL, newest Date first) via ExecuteSQL, so all five always agree.
// ALL five fields stay UNSTORED ("Do not store calculation results" checked) and keep
// their existing result types. Paste order does not matter; WeightLost,
// PercentWeightLossDisplay and TimeSinceLastWeighIn reference the fixed fields.
// Empty (not zero) when the patient has no weigh-in rows with an actual weight.


// ============================================================
// 1) CurrentWeight  (field id 191, Number, unstored)
//    was: Last ( WeighIn::Weight )
// ============================================================
Let ( [
~w = ExecuteSQL (
 "SELECT \"Weight\" FROM \"WeighIn\"
  WHERE \"AccountNumber\" = ? AND \"Weight\" IS NOT NULL AND \"Date\" IS NOT NULL
  ORDER BY \"Date\" DESC FETCH FIRST 1 ROWS ONLY" ; "" ; "" ; AccountNumber )
] ;
If ( not IsEmpty ( ~w ) and ~w ≠ "?" ; GetAsNumber ( ~w ) )
)


// ============================================================
// 2) TimeSinceLastWeighInDisplay  (field id 129, Number, unstored)
//    was: Get(CurrentDate) - GetNthRecord(WeighIn::Date; Count(WeighIn::Date) - 1)
// ============================================================
Let ( [
~iso = ExecuteSQL (
 "SELECT \"Date\" FROM \"WeighIn\"
  WHERE \"AccountNumber\" = ? AND \"Weight\" IS NOT NULL AND \"Date\" IS NOT NULL
  ORDER BY \"Date\" DESC FETCH FIRST 1 ROWS ONLY" ; "" ; "" ; AccountNumber )
] ;
If ( not IsEmpty ( ~iso ) and ~iso ≠ "?" ;
 Get ( CurrentDate ) - Date ( Middle ( ~iso ; 6 ; 2 ) ; Middle ( ~iso ; 9 ; 2 ) ; Left ( ~iso ; 4 ) ) )
)


// ============================================================
// 3) WeightLost  (field id 196, Number, unstored)
//    was: Weightpounds - Last ( WeighIn::Weight )
//    now leans on the fixed CurrentWeight - one source of truth
// ============================================================
If ( not IsEmpty ( CurrentWeight ) and Weightpounds > 0 ; Weightpounds - CurrentWeight )


// ============================================================
// 4) PercentWeightLossDisplay  (field id 143, Number, unstored)
//    was: If(IsEmpty(TimeSinceLastWeighInDisplay)=False;
//         Truncate(((Weightpounds-Last(WeighIn::Weight))/Weightpounds)*100;1))
// ============================================================
If ( not IsEmpty ( CurrentWeight ) and Weightpounds > 0 ;
 Truncate ( ( ( Weightpounds - CurrentWeight ) / Weightpounds ) * 100 ; 1 ) )


// ============================================================
// 5) TimeSinceLastWeighIn  (field id 128, Text, unstored) - BONUS
//    was: six repeated GetNthRecord(Count-1) fetches formatted as
//    days / weeks / years+months; now formats the fixed field #2
// ============================================================
Let ( d = TimeSinceLastWeighInDisplay ;
Case (
 IsEmpty ( d ) ; "" ;
 d < 14 ; d & " days" ;
 d < 365 ; Round ( d / 7 ; 1 ) & " weeks" ;
 Int ( d / 365 ) & " years " & Int ( Mod ( d ; 365 ) / 30 ) & " months"
) )


// ################ BATCH 2 - 2026-10-03 (same session) ################
// Three more main-layout fields off the positional pattern, found by sweeping
// the fresh Oct 3 DDR. Pasted + verified same day. Left as-is intentionally:
// Last(Visits::VisitSummary)/LastAOMRx/ProblemList FIB-4/returntoclinic fetches
// ("most recent non-empty entry" on Visits is the wanted semantics; Visits rows
// are created chronologically and have no placeholders) and the SECA body-comp
// family (scans entered complete; optional cleanup later). HighChartHTML2
// (legacy Highcharts-CDN chart, unused) marked for retirement instead of repair.


// ============================================================
// 6) CurrentBMI  (field id 192, Number, unstored)
//    was: Last ( WeighIn::BMI )
// ============================================================
Let ( [
~b = ExecuteSQL (
 "SELECT \"BMI\" FROM \"WeighIn\"
  WHERE \"AccountNumber\" = ? AND \"BMI\" IS NOT NULL AND \"Date\" IS NOT NULL
  ORDER BY \"Date\" DESC FETCH FIRST 1 ROWS ONLY" ; "" ; "" ; AccountNumber )
] ;
If ( not IsEmpty ( ~b ) and ~b ≠ "?" ; GetAsNumber ( ~b ) )
)


// ============================================================
// 7) LastWeight  (field id 130, Text, unstored)
//    was: GetNthRecord ( WeighIn::Weight ; Count ( WeighIn::Weight ) + 1 ) & " lbs"
//    (a Count+1 hack). Now formats the fixed CurrentWeight; falls back to the
//    enrolled weight when no weigh-ins exist. Semantic change: with exactly one
//    weigh-in it now shows that weigh-in rather than Weightpounds.
// ============================================================
If ( IsEmpty ( CurrentWeight ) ; Weightpounds & " lbs" ; CurrentWeight & " lbs" )


// ============================================================
// 8) lastVisit  (field id 189, Date, unstored)
//    was: If ( IsEmpty ( Last ( Visits::Date ) ) ; DateOfFirstVisit ; Last ( Visits::Date ) )
//    Feeds MochiRefillMsg's 90-day cadence; MAX(Date) is order-proof against
//    back-entered visits.
// ============================================================
Let ( [
~iso = ExecuteSQL ( "SELECT MAX(\"Date\") FROM \"Visits\" WHERE \"AccountNumber\" = ? AND \"Date\" IS NOT NULL" ; "" ; "" ; AccountNumber )
] ;
If ( IsEmpty ( ~iso ) or ~iso = "?" ; DateOfFirstVisit ;
 Date ( Middle ( ~iso ; 6 ; 2 ) ; Middle ( ~iso ; 9 ; 2 ) ; Left ( ~iso ; 4 ) ) )
)
