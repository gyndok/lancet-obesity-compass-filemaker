// MetabolicFlags v2.2 - 2026-10-01 - HOMA-IR + FIB-4 computed from Labsbefore text.
// Unstored Text calc on Obesity Management. Parses lines "YYYY-MM-DD (emdash) Name (emdash) Value ..."
// HOMA-IR = glucose*insulin/405. FIB-4 = age*AST/(platelets*sqrt(ALT)), age at the AST lab date.
// Tiers: HOMA <1 optimal / <2 normal / <3 borderline / >=3 abnormal.
// FIB-4 <1.3 normal (age>=65 uses <2.0) / <=2.67 indeterminate / >2.67 abnormal.
// v2.2: HOMA-IR now requires a SAME-DAY glucose + insulin pair (most recent date
//   holding both), matching the JS dashboard. Mixed-date pairs (e.g. 2026 glucose
//   x 2024 insulin) are no longer multiplied; instead an explanation line prints:
//   "HOMA-IR: not computed (... different draws - redraw fasting pair)".
//   Verified in production 2026-10-01 (patient w/ 2026 glucose + 2024 insulin).
// v2.1: leading-number value parse (fixes 246x10E9 -> 246109), per-analyte sanity
//   bounds (glucose 40-600, insulin 0.5-300, AST/ALT 3-3000, platelets 20-1000),
//   LOINC-style long-name matching (urine glucose / insulin antibody excluded).
Let ( [
src = Substitute ( Labsbefore ; Char(8212) ; "|" ) ;
n = ValueCount ( src ) ;

// ---- ALL qualifying glucose values, as "YYYY-MM-DD~value" lines ----
gAll = While ( [ i = 1 ; out = "" ] ; i <= n ; [
   ln = GetValue ( src ; i ) ;
   p1 = Position ( ln ; "|" ; 1 ; 1 ) ;
   p2 = Position ( ln ; "|" ; 1 ; 2 ) ;
   ok = Middle ( ln ; 5 ; 1 ) = "-" and Middle ( ln ; 8 ; 1 ) = "-" and p2 > 0 ;
   nm = If ( ok ; Trim ( Middle ( ln ; p1 + 1 ; p2 - p1 - 1 ) ) ) ;
   seg = If ( ok ; Trim ( Middle ( ln ; p2 + 1 ; 200 ) ) ) ;
   v = If ( ok ; GetAsNumber ( While ( [ j = 1 ; o2 = "" ; go = 1 ] ; go and j <= Length ( seg ) ; [
       c = Middle ( seg ; j ; 1 ) ;
       isn = ( c >= "0" and c <= "9" ) or c = "." ;
       o2 = If ( isn ; o2 & c ; o2 ) ;
       go = isn ;
       j = j + 1 ] ; o2 ) ) ) ;
   d = Left ( ln ; 10 ) ;
   hit = ok and ( nm = "Glucose" or Left ( nm ; 8 ) = "Glucose " )
     and Position ( nm ; "urine" ; 1 ; 1 ) = 0
     and not IsEmpty ( v ) and v >= 40 and v <= 600 ;
   out = If ( hit ; out & If ( out ≠ "" ; ¶ ) & d & "~" & v ; out ) ;
   i = i + 1 ] ; out ) ;

// ---- ALL qualifying insulin values ----
insAll = While ( [ i = 1 ; out = "" ] ; i <= n ; [
   ln = GetValue ( src ; i ) ;
   p1 = Position ( ln ; "|" ; 1 ; 1 ) ;
   p2 = Position ( ln ; "|" ; 1 ; 2 ) ;
   ok = Middle ( ln ; 5 ; 1 ) = "-" and Middle ( ln ; 8 ; 1 ) = "-" and p2 > 0 ;
   nm = If ( ok ; Trim ( Middle ( ln ; p1 + 1 ; p2 - p1 - 1 ) ) ) ;
   seg = If ( ok ; Trim ( Middle ( ln ; p2 + 1 ; 200 ) ) ) ;
   v = If ( ok ; GetAsNumber ( While ( [ j = 1 ; o2 = "" ; go = 1 ] ; go and j <= Length ( seg ) ; [
       c = Middle ( seg ; j ; 1 ) ;
       isn = ( c >= "0" and c <= "9" ) or c = "." ;
       o2 = If ( isn ; o2 & c ; o2 ) ;
       go = isn ;
       j = j + 1 ] ; o2 ) ) ) ;
   d = Left ( ln ; 10 ) ;
   hit = ok and ( nm = "Insulin" or Left ( nm ; 8 ) = "Insulin " )
     and Position ( nm ; "antibod" ; 1 ; 1 ) = 0
     and Position ( nm ; "resistance" ; 1 ; 1 ) = 0
     and not IsEmpty ( v ) and v >= .5 and v <= 300 ;
   out = If ( hit ; out & If ( out ≠ "" ; ¶ ) & d & "~" & v ; out ) ;
   i = i + 1 ] ; out ) ;

// ---- newest glucose / insulin independently (for the explanation line) ----
gNew = While ( [ k = 1 ; m = ValueCount ( gAll ) ; bD = "" ; bV = "" ] ; k <= m ; [
   e = GetValue ( gAll ; k ) ;
   d2 = Left ( e ; 10 ) ;
   good = d2 > bD ;
   bD = If ( good ; d2 ; bD ) ;
   bV = If ( good ; Middle ( e ; 12 ; 30 ) ; bV ) ;
   k = k + 1 ] ; bD & "~" & bV ) ;
iNew = While ( [ k = 1 ; m = ValueCount ( insAll ) ; bD = "" ; bV = "" ] ; k <= m ; [
   e = GetValue ( insAll ; k ) ;
   d2 = Left ( e ; 10 ) ;
   good = d2 > bD ;
   bD = If ( good ; d2 ; bD ) ;
   bV = If ( good ; Middle ( e ; 12 ; 30 ) ; bV ) ;
   k = k + 1 ] ; bD & "~" & bV ) ;

// ---- SAME-DAY pair: most recent date present in BOTH lists ----
pair = While ( [ k = 1 ; m = ValueCount ( gAll ) ; bD = "" ; bG = "" ] ; k <= m ; [
   e = GetValue ( gAll ; k ) ;
   d2 = Left ( e ; 10 ) ;
   has = Position ( ¶ & insAll & ¶ ; ¶ & d2 & "~" ; 1 ; 1 ) > 0 ;
   good = has and d2 > bD ;
   bD = If ( good ; d2 ; bD ) ;
   bG = If ( good ; Middle ( e ; 12 ; 30 ) ; bG ) ;
   k = k + 1 ] ; bD & "~" & bG ) ;
pairD = Left ( pair ; 10 ) ;
pairG = GetAsNumber ( Middle ( pair ; 12 ; 30 ) ) ;
pairI = If ( not IsEmpty ( pairD ) ;
   GetAsNumber ( Middle (
     GetValue ( Middle ( ¶ & insAll ; Position ( ¶ & insAll & ¶ ; ¶ & pairD & "~" ; 1 ; 1 ) + 1 ; 500 ) ; 1 ) ;
     12 ; 30 ) ) ) ;

gD = GetValue ( Substitute ( gNew ; "~" ; ¶ ) ; 1 ) ;
gV = GetAsNumber ( GetValue ( Substitute ( gNew ; "~" ; ¶ ) ; 2 ) ) ;
iD = GetValue ( Substitute ( iNew ; "~" ; ¶ ) ; 1 ) ;
iV = GetAsNumber ( GetValue ( Substitute ( iNew ; "~" ; ¶ ) ; 2 ) ) ;

ast = While ( [ i = 1 ; bd = "" ; bv = "" ] ; i <= n ; [
   ln = GetValue ( src ; i ) ;
   p1 = Position ( ln ; "|" ; 1 ; 1 ) ;
   p2 = Position ( ln ; "|" ; 1 ; 2 ) ;
   ok = Middle ( ln ; 5 ; 1 ) = "-" and Middle ( ln ; 8 ; 1 ) = "-" and p2 > 0 ;
   nm = If ( ok ; Trim ( Middle ( ln ; p1 + 1 ; p2 - p1 - 1 ) ) ) ;
   seg = If ( ok ; Trim ( Middle ( ln ; p2 + 1 ; 200 ) ) ) ;
   v = If ( ok ; GetAsNumber ( While ( [ j = 1 ; o2 = "" ; go = 1 ] ; go and j <= Length ( seg ) ; [
       c = Middle ( seg ; j ; 1 ) ;
       isn = ( c >= "0" and c <= "9" ) or c = "." ;
       o2 = If ( isn ; o2 & c ; o2 ) ;
       go = isn ;
       j = j + 1 ] ; o2 ) ) ) ;
   d = Left ( ln ; 10 ) ;
   hit = ok and ( nm = "AST (SGOT)" or nm = "AST" or Left ( nm ; 9 ) = "Aspartate" )
     and not IsEmpty ( v ) and v >= 3 and v <= 3000 and d > bd ;
   bd = If ( hit ; d ; bd ) ;
   bv = If ( hit ; v ; bv ) ;
   i = i + 1 ] ; bd & "|" & bv ) ;

alt = While ( [ i = 1 ; bd = "" ; bv = "" ] ; i <= n ; [
   ln = GetValue ( src ; i ) ;
   p1 = Position ( ln ; "|" ; 1 ; 1 ) ;
   p2 = Position ( ln ; "|" ; 1 ; 2 ) ;
   ok = Middle ( ln ; 5 ; 1 ) = "-" and Middle ( ln ; 8 ; 1 ) = "-" and p2 > 0 ;
   nm = If ( ok ; Trim ( Middle ( ln ; p1 + 1 ; p2 - p1 - 1 ) ) ) ;
   seg = If ( ok ; Trim ( Middle ( ln ; p2 + 1 ; 200 ) ) ) ;
   v = If ( ok ; GetAsNumber ( While ( [ j = 1 ; o2 = "" ; go = 1 ] ; go and j <= Length ( seg ) ; [
       c = Middle ( seg ; j ; 1 ) ;
       isn = ( c >= "0" and c <= "9" ) or c = "." ;
       o2 = If ( isn ; o2 & c ; o2 ) ;
       go = isn ;
       j = j + 1 ] ; o2 ) ) ) ;
   d = Left ( ln ; 10 ) ;
   hit = ok and ( nm = "ALT (SGPT)" or nm = "ALT" or Left ( nm ; 7 ) = "Alanine" )
     and not IsEmpty ( v ) and v >= 3 and v <= 3000 and d > bd ;
   bd = If ( hit ; d ; bd ) ;
   bv = If ( hit ; v ; bv ) ;
   i = i + 1 ] ; bd & "|" & bv ) ;

plt = While ( [ i = 1 ; bd = "" ; bv = "" ] ; i <= n ; [
   ln = GetValue ( src ; i ) ;
   p1 = Position ( ln ; "|" ; 1 ; 1 ) ;
   p2 = Position ( ln ; "|" ; 1 ; 2 ) ;
   ok = Middle ( ln ; 5 ; 1 ) = "-" and Middle ( ln ; 8 ; 1 ) = "-" and p2 > 0 ;
   nm = If ( ok ; Trim ( Middle ( ln ; p1 + 1 ; p2 - p1 - 1 ) ) ) ;
   seg = If ( ok ; Trim ( Middle ( ln ; p2 + 1 ; 200 ) ) ) ;
   v = If ( ok ; GetAsNumber ( While ( [ j = 1 ; o2 = "" ; go = 1 ] ; go and j <= Length ( seg ) ; [
       c = Middle ( seg ; j ; 1 ) ;
       isn = ( c >= "0" and c <= "9" ) or c = "." ;
       o2 = If ( isn ; o2 & c ; o2 ) ;
       go = isn ;
       j = j + 1 ] ; o2 ) ) ) ;
   d = Left ( ln ; 10 ) ;
   hit = ok and ( nm = "Platelets" or nm = "Platelet Count" )
     and not IsEmpty ( v ) and v >= 20 and v <= 1000 and d > bd ;
   bd = If ( hit ; d ; bd ) ;
   bv = If ( hit ; v ; bv ) ;
   i = i + 1 ] ; bd & "|" & bv ) ;

aD = GetValue ( Substitute ( ast ; "|" ; ¶ ) ; 1 ) ;
aV = GetAsNumber ( GetValue ( Substitute ( ast ; "|" ; ¶ ) ; 2 ) ) ;
lV = GetAsNumber ( GetValue ( Substitute ( alt ; "|" ; ¶ ) ; 2 ) ) ;
pV = GetAsNumber ( GetValue ( Substitute ( plt ; "|" ; ¶ ) ; 2 ) ) ;

homa = If ( pairG > 0 and pairI > 0 ; Truncate ( pairG * pairI / 405 ; 2 ) ) ;
astDate = If ( not IsEmpty ( aD ) ;
  Date ( Middle ( aD ; 6 ; 2 ) ; Middle ( aD ; 9 ; 2 ) ; Left ( aD ; 4 ) ) ) ;
age = If ( not IsEmpty ( PT_DOB ) and not IsEmpty ( astDate ) ;
  Year ( astDate ) - Year ( PT_DOB ) -
  ( astDate < Date ( Month ( PT_DOB ) ; Day ( PT_DOB ) ; Year ( astDate ) ) ) ) ;
fib = If ( aV > 0 and lV > 0 and pV > 0 and age > 0 ;
  Round ( age * aV / ( pV * Sqrt ( lV ) ) ; 2 ) ) ;

hFlag = Case (
  homa < 1 ; "NORMAL (optimal insulin sensitivity)" ;
  homa < 2 ; "NORMAL (healthy insulin sensitivity)" ;
  homa < 3 ; "BORDERLINE (early insulin resistance)" ;
  "ABNORMAL (significant insulin resistance)" ) ;
fLow = If ( age >= 65 ; 2 ; 1.3 ) ;
fFlag = Case (
  fib < fLow ; "NORMAL (low risk of advanced fibrosis" & If ( age >= 65 ; ", age-adjusted cutoff 2.0" ; "" ) & ")" ;
  fib <= 2.67 ; "INDETERMINATE (consider elastography)" ;
  "ABNORMAL (high risk - consider hepatology referral)" ) ;

hLine = Case (
  not IsEmpty ( homa ) ;
    "HOMA-IR: " & homa & " (" & pairD & ") - " & hFlag ;
  gV > 0 and iV > 0 ;
    "HOMA-IR: not computed (glucose " & gD & " and insulin " & iD & " from different draws - redraw fasting pair)" ;
  "" ) ;
fLine = If ( not IsEmpty ( fib ) ; "FIB-4: " & fib & " (" & aD & ") - " & fFlag )
] ;
If ( IsEmpty ( hLine ) and IsEmpty ( fLine ) ; "" ;
  hLine & If ( not IsEmpty ( hLine ) and not IsEmpty ( fLine ) ; Char(13) ) & fLine )
)
